     1	"""Validated boundary for injecting precomputed transport into RealWonder."""
     2	
     3	from __future__ import annotations
     4	
     5	from pathlib import Path
     6	from typing import Literal
     7	
     8	import torch
     9	
    10	
    11	TransportCondition = Literal["correct", "shuffled", "flow", "blend"]
    12	
    13	_LATENT_KEYS = {
    14	    "correct": "correct_fused_latent",
    15	    "shuffled": "shuffled_fused_latent",
    16	    "flow": "flow_fused_latent",
    17	    "blend": "blend_fused_latent",
    18	}
    19	
    20	
    21	def load_precomputed_transport_latent(
    22	    artifact_path: str | Path,
    23	    *,
    24	    mode: TransportCondition,
    25	    reference_latent: torch.Tensor,
    26	) -> torch.Tensor:
    27	    """Load a fused latent and match the existing RealWonder ``sim_latent``.
    28	
    29	    The artifact is produced by ``run_wan_vae_transport_probe.py``. Requiring
    30	    exact agreement with the freshly encoded reference prevents an artifact
    31	    from a different case, frame count, resolution, or latent layout from
    32	    silently entering the diffusion pipeline.
    33	    """
    34	
    35	    if mode not in _LATENT_KEYS:
    36	        raise ValueError(f"unsupported transport mode: {mode}")
    37	    if reference_latent.ndim != 5:
    38	        raise ValueError("reference_latent must have shape [B,T,C,H,W]")
    39	
    40	    path = Path(artifact_path)
    41	    if not path.is_file():
    42	        raise FileNotFoundError(path)
    43	    state = torch.load(path, map_location="cpu", weights_only=True)
    44	    if not isinstance(state, dict):
    45	        raise ValueError("transport artifact must contain a dictionary")
    46	
    47	    key = _LATENT_KEYS[mode]
    48	    selected = state.get(key)
    49	    if not isinstance(selected, torch.Tensor):
    50	        raise ValueError(f"transport artifact is missing tensor {key!r}")
    51	    if selected.ndim != 5 or tuple(selected.shape) != tuple(reference_latent.shape):
    52	        raise ValueError(
    53	            f"{key} shape {tuple(selected.shape)} does not match freshly encoded "
    54	            f"sim_latent {tuple(reference_latent.shape)}"
    55	        )
    56	    if not selected.dtype.is_floating_point or not bool(torch.isfinite(selected).all()):
    57	        raise ValueError(f"{key} must be a finite floating-point tensor")
    58	
    59	    frame_count, height, width = selected.shape[1], selected.shape[3], selected.shape[4]
    60	    mask = state.get("transport_mask")
    61	    count = state.get("contribution_count")
    62	    if not isinstance(mask, torch.Tensor) or tuple(mask.shape) != (
    63	        frame_count,
    64	        1,
    65	        height,
    66	        width,
    67	    ):
    68	        raise ValueError("transport_mask does not match the selected latent")
    69	    if mask.dtype != torch.bool:
    70	        raise ValueError("transport_mask must be boolean")
    71	    if not isinstance(count, torch.Tensor) or tuple(count.shape) != tuple(mask.shape):
    72	        raise ValueError("contribution_count does not match transport_mask")
    73	    if count.dtype.is_floating_point or bool((count < 0).any()):
    74	        raise ValueError("contribution_count must contain nonnegative integers")
    75	    if not torch.equal(mask, count > 0):
    76	        raise ValueError("transport_mask must equal contribution_count > 0")
    77	
    78	    return selected.to(
    79	        device=reference_latent.device,
    80	        dtype=reference_latent.dtype,
    81	    ).contiguous()
