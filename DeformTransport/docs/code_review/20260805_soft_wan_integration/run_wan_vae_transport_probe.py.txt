    59	def _sha256(path: Path) -> str:
    60	    digest = hashlib.sha256()
    61	    with path.open("rb") as file:
    62	        for chunk in iter(lambda: file.read(4 * 1024 * 1024), b""):
    63	            digest.update(chunk)
    64	    return digest.hexdigest()

    67	def _gpu_used_memory_mib() -> int | None:
    68	    visible_devices = os.environ.get("CUDA_VISIBLE_DEVICES", "").strip()
    69	    if visible_devices == "-1":
    70	        return None
    71	
    72	    gpu_id = visible_devices.split(",", maxsplit=1)[0].strip() or "0"
    73	
    74	    try:
    75	        result = subprocess.run(
    76	            [
    77	                "nvidia-smi",
    78	                "--query-gpu=memory.used",
    79	                "--format=csv,noheader,nounits",
    80	                f"--id={gpu_id}",
    81	            ],
    82	            check=True,
    83	            capture_output=True,
    84	            text=True,
    85	            timeout=10,
    86	        )
    87	        return int(result.stdout.strip().splitlines()[0])
    88	    except (FileNotFoundError, IndexError, subprocess.SubprocessError, ValueError):
    89	        return None

    92	def _timed_cuda(function):
    93	    torch.cuda.synchronize()
    94	    started = time.perf_counter()
    95	    output = function()
    96	    torch.cuda.synchronize()
    97	    return output, time.perf_counter() - started

   109	def _uint8_frames(frames: torch.Tensor) -> np.ndarray:
   110	    return (
   111	        frames.detach()
   112	        .cpu()
   113	        .clamp(0.0, 1.0)
   114	        .permute(0, 2, 3, 1)
   115	        .mul(255.0)
   116	        .round()
   117	        .to(torch.uint8)
   118	        .numpy()
   119	    )

   131	def _source_comparison(source: torch.Tensor, reconstruction: torch.Tensor) -> Image.Image:
   132	    source_image = Image.fromarray(_uint8_frames(source)[0])
   133	    reconstruction_image = Image.fromarray(_uint8_frames(reconstruction)[0])
   134	    canvas = Image.new("RGB", (1664, 512), "black")
   135	    for panel_index, (label, image) in enumerate(
   136	        (("Input source RGB", source_image), ("Wan VAE reconstruction", reconstruction_image))
   137	    ):
   138	        resized = image.resize((832, 480), resample=Image.Resampling.BILINEAR)
   139	        left = panel_index * 832
   140	        canvas.paste(resized, (left, 32))
   141	        ImageDraw.Draw(canvas).text((left + 10, 9), label, fill="white")
   142	    return canvas

   145	def _labeled_four_panel(frames_by_label: list[tuple[str, np.ndarray]]) -> np.ndarray:
   146	    outputs = []
   147	    frame_count = len(frames_by_label[0][1])
   148	    for frame_index in range(frame_count):
   149	        panels = []
   150	        for label, frames in frames_by_label:
   151	            image = Image.fromarray(frames[frame_index]).resize(
   152	                (416, 240), resample=Image.Resampling.BILINEAR
   153	            )
   154	            panel = Image.new("RGB", (416, 272), "black")
   155	            panel.paste(image, (0, 32))
   156	            ImageDraw.Draw(panel).text((8, 9), label, fill="white")
   157	            panels.append(np.asarray(panel))
   158	        outputs.append(np.concatenate(panels, axis=1))
   159	    return np.stack(outputs)

   162	def _expand_latent_mask_to_pixel_frames(mask: torch.Tensor, pixel_frames: int) -> torch.Tensor:
   163	    expanded = [mask[:1]]
   164	    for latent_index in range(1, mask.shape[0]):
   165	        expanded.append(mask[latent_index : latent_index + 1].repeat(4, 1, 1, 1))
   166	    expanded_mask = torch.cat(expanded, dim=0)
   167	    return expanded_mask[:pixel_frames]

   170	def _latent_frame_metrics(prediction, target, mask):
   171	    metrics = []
   172	    for frame_index in range(prediction.shape[0]):
   173	        expanded = mask[frame_index].expand(prediction.shape[1], -1, -1)
   174	        selected = (prediction[frame_index] - target[frame_index])[expanded]
   175	        metrics.append(
   176	            {
   177	                "frame": frame_index,
   178	                "support_cells": int(mask[frame_index].sum()),
   179	                "masked_l1": float(selected.abs().mean()),
   180	                "masked_mse": float(selected.square().mean()),
   181	                "masked_psnr_db": None,
   182	            }
   183	        )
   184	    return metrics

   187	def _tensor_statistics(tensor: torch.Tensor) -> dict:
   188	    tensor = tensor.detach().float()
   189	    return {
   190	        "shape": list(tensor.shape),
   191	        "dtype": str(tensor.dtype),
   192	        "mean": float(tensor.mean()),
   193	        "std": float(tensor.std()),
   194	        "min": float(tensor.min()),
   195	        "max": float(tensor.max()),
   196	        "l2_norm": float(torch.linalg.vector_norm(tensor)),
   197	        "finite": bool(torch.isfinite(tensor).all()),
   198	    }

   357	        fused_shuffled_latent

   362	    raw_correct_01 = _pixels_01(raw_correct_decoded)
   363	    raw_shuffled_01 = _pixels_01(raw_shuffled_decoded)

   441	    }
   442	    latent_outputs_path = output_dir / "vae_latent_outputs.pt"

   460	    videos = {
   461	        "target_input": _uint8_frames(target_pixels_01),
   462	        "original_reconstruction": _uint8_frames(original_01),
   463	        "raw_correct": _uint8_frames(raw_correct_01),

   492	    latent_different = not torch.equal(raw_correct_latent, raw_shuffled_latent)
   493	    pixel_frame_count = int(future_pixels.shape[2])
   494	    expected_latent_slots = int(latent_frame_indices.numel())
   495	    expected_future_latent_shape = [
   496	        1,
   497	        expected_latent_slots,
   498	        *list(source_latent.shape[2:]),

   501	    engineering_checks = {
