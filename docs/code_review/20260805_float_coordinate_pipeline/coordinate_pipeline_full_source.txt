    20	def _plain_cpu_tensor(value: Any, *, dtype: torch.dtype) -> torch.Tensor:
    21	    """Detach runtime tensor subclasses at the trajectory export boundary."""
    22	    tensor = torch.as_tensor(value)
    23	    if type(tensor) is not torch.Tensor:

    31	def _remove_singleton_batch(tensor: torch.Tensor, expected_dims: int) -> torch.Tensor:
    32	    if tensor.ndim == expected_dims + 1 and tensor.shape[0] == 1:
    33	        return tensor[0]
    34	    return tensor

    37	def project_points_realwonder(
    38	    points_xyz: torch.Tensor,
    39	    K: torch.Tensor,
    40	    R: torch.Tensor,
    41	    T: torch.Tensor,
    42	    image_size: int = 512,
    43	) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    44	    """Project PyTorch3D-space points exactly as RealWonder does.
    45	
    46	    Returns ``(uv, depth, valid)`` where ``uv`` is in RealWonder's 512-square
    47	    render coordinates and ``valid`` only tests positive depth and image bounds.
    48	    Occlusion is deliberately left to the renderer.
    49	    """
    50	
    51	    points_xyz = torch.as_tensor(points_xyz)
    52	    K = _remove_singleton_batch(torch.as_tensor(K), 2)
    53	    R = _remove_singleton_batch(torch.as_tensor(R), 2)
    54	    T = _remove_singleton_batch(torch.as_tensor(T), 1)
    55	
    56	    if points_xyz.ndim != 2 or points_xyz.shape[-1] != 3:
    57	        raise ValueError(f"points_xyz must have shape [N, 3], got {tuple(points_xyz.shape)}")
    58	    if K.shape not in ((3, 3), (4, 4)):
    59	        raise ValueError(f"K must have shape [3, 3] or [4, 4], got {tuple(K.shape)}")
    60	    if R.shape != (3, 3):
    61	        raise ValueError(f"R must have shape [3, 3], got {tuple(R.shape)}")
    62	    if T.shape != (3,):
    63	        raise ValueError(f"T must have shape [3], got {tuple(T.shape)}")
    64	
    65	    device = points_xyz.device
    66	    dtype = points_xyz.dtype
    67	    intrinsics = K[:3, :3].clone().to(device=device, dtype=dtype)
    68	    rotation = R.to(device=device, dtype=dtype)
    69	    translation = T.to(device=device, dtype=dtype)
    70	
    71	    # RealWonder receives PyTorch3D's 4x4 calibration matrix, whose [2, 2]
    72	    # element is zero, and explicitly restores it before projection.
    73	    intrinsics[2, 2] = 1.0
    74	    camera_xyz = (rotation @ points_xyz.T).T + translation
    75	    image_xyz = (intrinsics @ camera_xyz.T).T
    76	    depth = camera_xyz[:, 2]
    77	    uv = image_xyz[:, :2] / image_xyz[:, 2:3].clamp_min(1e-3)
    78	    uv = float(image_size) - uv
    79	
    80	    finite = torch.isfinite(points_xyz).all(dim=1) & torch.isfinite(uv).all(dim=1)
    81	    valid = (
    82	        finite
    83	        & (depth > 1e-3)
    84	        & (uv[:, 0] >= 0)
    85	        & (uv[:, 0] < image_size)
    86	        & (uv[:, 1] >= 0)
    87	        & (uv[:, 1] < image_size)
    88	    )
    89	    return uv, depth, valid

    92	def map_image_uv_to_latent(
    93	    uv: torch.Tensor,
    94	    *,
    95	    source_size: int = 512,
    96	    resized_size: int = 832,
    97	    crop_start: int = 176,
    98	    output_height: int = 480,
    99	    latent_stride: int = 8,
   100	) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
   101	    """Map RealWonder render coordinates into its cropped Wan latent grid.
   102	
   103	    The offline pipeline first resizes 512x512 to 832x832, takes a 480x832
   104	    vertical crop, and spatially downsamples by eight to a 60x104 latent grid.
   105	    """
   106	
   107	    uv = torch.as_tensor(uv)
   108	    if uv.ndim != 2 or uv.shape[-1] != 2:
   109	        raise ValueError(f"uv must have shape [N, 2], got {tuple(uv.shape)}")
   110	
   111	    scale = float(resized_size) / float(source_size)
   112	    video_uv = uv * scale
   113	    video_uv = video_uv.clone()
   114	    video_uv[:, 1] -= float(crop_start)
   115	    valid = (
   116	        torch.isfinite(video_uv).all(dim=1)
   117	        & (video_uv[:, 0] >= 0)
   118	        & (video_uv[:, 0] < resized_size)
   119	        & (video_uv[:, 1] >= 0)
   120	        & (video_uv[:, 1] < output_height)
   121	    )
   122	
   123	    latent_xy = torch.floor(video_uv / float(latent_stride)).to(torch.long)
   124	    return latent_xy, video_uv, valid

   246	                    "object_id": object_id,
   247	                    "material_type": self.material_types[object_id],
   248	                    "initial_points_3d": initial_points,
   249	                    "initial_points_uv": initial_uv,
   250	                    "initial_depth": initial_depth,

   283	def validate_trajectory_export(state: Mapping[str, Any]) -> None:
   284	    """Raise a descriptive error when an export breaks the stage-1 contract."""
   285	
   286	    if state.get("format_version") != TRAJECTORY_FORMAT_VERSION:
   287	        raise ValueError(f"unsupported trajectory format: {state.get('format_version')}")
   288	    frame_ids = state.get("frame_ids")
   289	    simulation_steps = state.get("simulation_steps")
   290	    objects = state.get("objects")
   291	    if not isinstance(frame_ids, torch.Tensor) or frame_ids.ndim != 1:
   292	        raise ValueError("frame_ids must be a 1D tensor")
   293	    if not isinstance(simulation_steps, torch.Tensor) or simulation_steps.shape != frame_ids.shape:
   294	        raise ValueError("simulation_steps must match frame_ids")
   295	    if frame_ids.numel() == 0:
   296	        raise ValueError("trajectory export has no frames")
   297	    if not isinstance(objects, list) or not objects:
   298	        raise ValueError("trajectory export has no objects")
   299	
   300	    expected_frames = frame_ids.numel()
   301	    for object_state in objects:
   302	        points_3d = object_state["points_3d"]
   303	        points_uv = object_state["points_uv"]
   304	        valid = object_state["projection_valid"]
   305	        initial = object_state["initial_points_3d"]
   306	        if points_3d.ndim != 3 or points_3d.shape[0] != expected_frames or points_3d.shape[2] != 3:
   307	            raise ValueError("points_3d must have shape [T, N, 3]")
   308	        if points_uv.shape != points_3d.shape[:2] + (2,):
   309	            raise ValueError("points_uv must have shape [T, N, 2]")
   310	        if valid.shape != points_3d.shape[:2] or valid.dtype != torch.bool:
   311	            raise ValueError("projection_valid must be bool with shape [T, N]")
   312	        if initial.shape != points_3d.shape[1:]:
   313	            raise ValueError("initial point identity does not match trajectory point identity")
   314	        if not torch.isfinite(points_3d).all() or not torch.isfinite(points_uv).all():
   315	            raise ValueError("trajectory export contains non-finite values")
   316	        bindings = object_state.get("binding_particle_indices")
   317	        if bindings is not None and (bindings.ndim != 2 or bindings.shape[0] != initial.shape[0]):
   318	            raise ValueError("binding_particle_indices must have shape [N, K]")

   183	    future_latent, future_video, future_crop_valid = _map_frames_to_latent(
   184	        flat["points_uv"]
   185	    )
   186	    source_valid = flat["source_render_valid"] & source_crop_valid
   187	    projection_valid = flat["render_valid"] & future_crop_valid
   188	    source_visible = torch.zeros(point_count, dtype=torch.bool)
   189	    source_visible[raster_visible_ids] = True
   190	    source_visible &= source_valid
   191	    source_visible_point_ids = point_id[source_visible]
   192	
   193	    if coarse_rgb_paths and len(coarse_rgb_paths) != frame_count:
   194	        raise ValueError(
   195	            f"expected {frame_count} coarse RGB paths, got {len(coarse_rgb_paths)}"
   196	        )
   197	
   198	    def path_string(path: str | Path | None) -> str | None:
   199	        return str(Path(path).resolve()) if path is not None else None
   200	
   201	    state = {
   202	        "format_version": TRANSPORT_READY_FORMAT_VERSION,
   203	        "case_name": str(case_name),
   204	        "coordinate_system": {
   205	            "points_3d": trajectory.get("coordinate_system"),
   206	            "points_2d_render": "realwonder_512_uv_xy",
   207	            "points_2d_video": "realwonder_resized_832_crop_480_uv_xy",
   208	            "points_2d_latent": "wan_spatial_cell_xy_nearest",
   209	        },
   210	        "frame_ids": _plain_cpu(trajectory["frame_ids"]).to(torch.long),
   211	        "simulation_steps": _plain_cpu(trajectory["simulation_steps"]).to(
   212	            torch.long
   213	        ),
   214	        "point_id": point_id,
   215	        "object_id": flat["object_id"],
   216	        "material_type": flat["material_type"],
   217	        "object_point_ranges": flat["object_point_ranges"],
   218	        "points_3d": flat["points_3d"],
   219	        "points_2d_render": flat["points_uv"],
   220	        "points_2d_video": future_video,
   221	        "points_2d_latent": future_latent,
   222	        "depth": flat["depth"],
   223	        "render_projection_valid": flat["render_valid"],
   224	        "projection_valid": projection_valid,
   225	        "source_points_3d": flat["source_points_3d"],
   226	        "source_points_2d_render": flat["source_points_uv"],
   227	        "source_points_2d_video": source_video,
   228	        "source_points_2d_latent": source_latent,
   229	        "source_depth": flat["source_depth"],
   230	        "source_render_projection_valid": flat["source_render_valid"],
   231	        "source_valid": source_valid,
   232	        "source_visible": source_visible,
   233	        "source_visible_point_ids": source_visible_point_ids,
   234	        "source_raster_visible_point_ids": raster_visible_ids,
   235	        "point_particle_binding": flat["point_particle_binding"],
   236	        "point_particle_binding_by_object": flat[
   237	            "point_particle_binding_by_object"
   238	        ],
   239	        "camera": _plain_structure(trajectory["camera"]),
   240	        "render_height": RENDER_HEIGHT,
   241	        "render_width": RENDER_WIDTH,
   242	        "video_height": VIDEO_HEIGHT,
   243	        "video_width": VIDEO_WIDTH,
   244	        "latent_height": LATENT_HEIGHT,
   245	        "latent_width": LATENT_WIDTH,
   246	        "paths": {
   247	            "source_trajectory": path_string(source_trajectory_path),
   248	            "initial_rgb": path_string(initial_rgb_path),
   249	            "coarse_rgb_frames": [path_string(path) for path in coarse_rgb_paths],
   250	            "flow": path_string(flow_path),
   251	            "source_raster_point_indices": path_string(source_raster_path),
   252	        },
   253	    }
   254	    state = _plain_structure(state)
   255	    validate_transport_ready(state)
   256	    return state

   259	def _require_tensor(
   260	    state: Mapping[str, Any],
   261	    name: str,
   262	    *,
   263	    shape: tuple[int, ...],
   264	    dtype: torch.dtype,
   265	) -> torch.Tensor:
   266	    value = state.get(name)
   267	    if type(value) is not torch.Tensor:
   268	        raise ValueError(f"{name} must be a plain torch.Tensor")
   269	    if tuple(value.shape) != shape or value.dtype != dtype:
   270	        raise ValueError(
   271	            f"{name} must be {dtype} with shape {shape}, got "
   272	            f"{value.dtype} {tuple(value.shape)}"
   273	        )
   274	    if value.device.type != "cpu":
   275	        raise ValueError(f"{name} must be stored on CPU")
   276	    return value

   279	def validate_transport_ready(state: Mapping[str, Any]) -> None:
   280	    """Validate the flattened transport contract and verified crop mapping."""
   281	
   282	    if state.get("format_version") != TRANSPORT_READY_FORMAT_VERSION:
   283	        raise ValueError(
   284	            f"unsupported transport-ready format: {state.get('format_version')}"
   285	        )
   286	    frame_ids = state.get("frame_ids")
   287	    point_id = state.get("point_id")
   288	    if type(frame_ids) is not torch.Tensor or frame_ids.ndim != 1:
   289	        raise ValueError("frame_ids must be a plain 1D tensor")
   290	    if type(point_id) is not torch.Tensor or point_id.ndim != 1:
   291	        raise ValueError("point_id must be a plain 1D tensor")
   292	    frame_count = int(frame_ids.numel())
   293	    point_count = int(point_id.numel())
   294	    if frame_count == 0 or point_count == 0:
   295	        raise ValueError("transport-ready export cannot be empty")
   296	    if not torch.equal(point_id, torch.arange(point_count, dtype=torch.long)):
   297	        raise ValueError("point_id must preserve contiguous flattened point order")
   298	
   299	    _require_tensor(
   300	        state, "simulation_steps", shape=(frame_count,), dtype=torch.long
   301	    )
   302	    _require_tensor(state, "object_id", shape=(point_count,), dtype=torch.long)
   303	    points_3d = _require_tensor(
   304	        state,
   305	        "points_3d",
   306	        shape=(frame_count, point_count, 3),
   307	        dtype=torch.float32,
   308	    )
   309	    render_uv = _require_tensor(
   310	        state,
   311	        "points_2d_render",
   312	        shape=(frame_count, point_count, 2),
   313	        dtype=torch.float32,
   314	    )
   315	    _require_tensor(
   316	        state,
   317	        "points_2d_video",
   318	        shape=(frame_count, point_count, 2),
   319	        dtype=torch.float32,
   320	    )
   321	    latent_uv = _require_tensor(
   322	        state,
   323	        "points_2d_latent",
   324	        shape=(frame_count, point_count, 2),
   325	        dtype=torch.long,
   326	    )
   327	    depth = _require_tensor(
   328	        state,
   329	        "depth",
   330	        shape=(frame_count, point_count),
   331	        dtype=torch.float32,
   332	    )
   333	    render_valid = _require_tensor(
   334	        state,
   335	        "render_projection_valid",
   336	        shape=(frame_count, point_count),
   337	        dtype=torch.bool,
   338	    )
   339	    projection_valid = _require_tensor(
   340	        state,
   341	        "projection_valid",
   342	        shape=(frame_count, point_count),
   343	        dtype=torch.bool,
   344	    )
   345	    source_points = _require_tensor(
   346	        state,
   347	        "source_points_3d",
   348	        shape=(point_count, 3),
   349	        dtype=torch.float32,
   350	    )
   351	    source_render_uv = _require_tensor(
   352	        state,
   353	        "source_points_2d_render",
   354	        shape=(point_count, 2),
   355	        dtype=torch.float32,
   356	    )
   357	    _require_tensor(
   358	        state,
   359	        "source_points_2d_video",
   360	        shape=(point_count, 2),
   361	        dtype=torch.float32,
   362	    )
   363	    source_latent_uv = _require_tensor(
   364	        state,
   365	        "source_points_2d_latent",
   366	        shape=(point_count, 2),
   367	        dtype=torch.long,
   368	    )
   369	    source_depth = _require_tensor(
   370	        state, "source_depth", shape=(point_count,), dtype=torch.float32
   371	    )
   372	    source_render_valid = _require_tensor(
   373	        state,
   374	        "source_render_projection_valid",
   375	        shape=(point_count,),
   376	        dtype=torch.bool,
   377	    )
   378	    source_valid = _require_tensor(
   379	        state, "source_valid", shape=(point_count,), dtype=torch.bool
   380	    )
   381	    source_visible = _require_tensor(
   382	        state, "source_visible", shape=(point_count,), dtype=torch.bool
   383	    )
   384	    source_visible_ids = state.get("source_visible_point_ids")
   385	    if type(source_visible_ids) is not torch.Tensor or source_visible_ids.dtype != torch.long:
   386	        raise ValueError("source_visible_point_ids must be a plain int64 tensor")
   387	    if source_visible_ids.ndim != 1:
   388	        raise ValueError("source_visible_point_ids must be one-dimensional")
   389	    if source_visible_ids.numel() and (
   390	        int(source_visible_ids.min()) < 0
   391	        or int(source_visible_ids.max()) >= point_count
   392	    ):
   393	        raise ValueError("source_visible_point_ids contains an out-of-range ID")
   394	    if not torch.equal(source_visible_ids, point_id[source_visible]):
   395	        raise ValueError("source_visible IDs and mask disagree")
   396	    if bool((source_visible & ~source_valid).any()):
   397	        raise ValueError("source_visible must be a subset of source_valid")
   398	
   399	    expected_source_latent, _, source_crop_valid = map_image_uv_to_latent(
   400	        source_render_uv
   401	    )
   402	    expected_future_latent, _, future_crop_valid = _map_frames_to_latent(render_uv)
   403	    if not torch.equal(source_latent_uv, expected_source_latent):
   404	        raise ValueError("source latent coordinates do not match the verified mapping")
   405	    if not torch.equal(latent_uv, expected_future_latent):
   406	        raise ValueError("future latent coordinates do not match the verified mapping")
   407	    if not torch.equal(source_valid, source_render_valid & source_crop_valid):
   408	        raise ValueError("source_valid does not combine render and crop validity")
   409	    if not torch.equal(projection_valid, render_valid & future_crop_valid):
   410	        raise ValueError("projection_valid does not combine render and crop validity")
   411	
   412	    valid_source_cells = source_latent_uv[source_valid]
   413	    valid_future_cells = latent_uv[projection_valid]
   414	    for cells, label in (
   415	        (valid_source_cells, "source"),
   416	        (valid_future_cells, "future"),
   417	    ):
   418	        if cells.numel() and not bool(
   419	            (cells[:, 0] >= 0).all()
   420	            and (cells[:, 0] < int(state["latent_width"])).all()
   421	            and (cells[:, 1] >= 0).all()
   422	            and (cells[:, 1] < int(state["latent_height"])).all()
   423	        ):
   424	            raise ValueError(f"valid {label} latent coordinates are out of bounds")
   425	
   426	    for tensor, name in (
   427	        (points_3d, "points_3d"),
   428	        (render_uv, "points_2d_render"),
   429	        (depth, "depth"),
   430	        (source_points, "source_points_3d"),
   431	        (source_render_uv, "source_points_2d_render"),
   432	        (source_depth, "source_depth"),
   433	    ):
   434	        if not bool(torch.isfinite(tensor).all()):
   435	            raise ValueError(f"{name} contains NaN or Inf")
   436	
   437	    binding = state.get("point_particle_binding")
   438	    if binding is not None:
   439	        if type(binding) is not torch.Tensor or binding.dtype != torch.long:
   440	            raise ValueError("point_particle_binding must be a plain int64 tensor")
   441	        if binding.ndim != 2 or binding.shape[0] != point_count:
   442	            raise ValueError("point_particle_binding must have shape [N, K]")
   443	
   444	    camera = state.get("camera")
   445	    if not isinstance(camera, dict) or set(camera) != {"K", "R", "T"}:
   446	        raise ValueError("camera must contain K, R, and T")
   447	    for name, tensor in camera.items():
   448	        if type(tensor) is not torch.Tensor or tensor.shape[0] != frame_count:
   449	            raise ValueError(f"camera {name} must be a plain per-frame tensor")
   450	
   451	    paths = state.get("paths")
   452	    if not isinstance(paths, dict):
   453	        raise ValueError("paths metadata is required")
   454	    coarse_paths = paths.get("coarse_rgb_frames")
   455	    if not isinstance(coarse_paths, list) or len(coarse_paths) not in (0, frame_count):
   456	        raise ValueError("coarse_rgb_frames must be empty or match frame count")
   457	
   458	
   459	def save_transport_ready(state: Mapping[str, Any], path: str | Path) -> Path:
   460	    """Validate and save a portable CPU-tensor transport artifact."""
   461	
   462	    validate_transport_ready(state)
   463	    path = Path(path)
   464	    path.parent.mkdir(parents=True, exist_ok=True)
   465	    torch.save(dict(state), path)
   466	    loaded = torch.load(path, map_location="cpu", weights_only=False)
   467	    validate_transport_ready(loaded)
   468	    return path
